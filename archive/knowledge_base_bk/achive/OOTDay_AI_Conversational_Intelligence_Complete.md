### **Tone Selector Logic**

```python
def select_tone(user_profile, context):
    """
    Select appropriate conversational tone
    """
    
    # Check user age
    if user_profile.age < 25:
        base_tone = "casual_friendly"
    elif user_profile.age > 40:
        base_tone = "polite_helpful"
    else:
        base_tone = "polite_helpful"  # Default safe
    
    # Check conversation history
    if user_profile.conversation_count > 5:
        # Built rapport, can be more casual
        if base_tone == "polite_helpful":
            base_tone = "friendly_polite"  # Mix
    
    # Check context
    if context == "frustration":
        return "empathetic"
    elif context == "success":
        return "excited"
    elif context == "warning":
        return "concerned"
    elif context == "first_message":
        return "polite_helpful"  # Always polite first time
    
    return base_tone
```

### **Emoji Insertion Logic**

```python
def add_emoji(text, sentiment, tone):
    """
    Add appropriate emoji to text
    """
    
    # Don't add to questions usually
    if text.endswith("?") and tone != "excited":
        return text
    
    # Don't add to warnings
    if tone == "concerned":
        if "ระวัง" in text or "อย่าลืม" in text:
            return text + " ⚠️"
        return text
    
    # Add based on sentiment
    if sentiment == "positive":
        emojis = ["😊", "✨", "💕"]
    elif sentiment == "very_positive":
        emojis = ["🎉", "😍", "💯", "🔥"]
    elif sentiment == "neutral":
        emojis = ["", ""]  # Often none
    
    # Select random but appropriate
    emoji = random.choice(emojis)
    
    # Max 1-2 per message
    if text.count("emoji_placeholder") < 2:
        return text + " " + emoji
    
    return text
```

## A. Vague Request Handling

### **Pattern Recognition**

```python
class VagueRequestDetector:
    """
    Detect and handle vague user requests
    """
    
    VAGUE_PATTERNS = [
        "หาชุด",
        "หาเสื้อ",
        "หากางเกง",
        "อยากได้ชุด",
        "ช่วยหา",
        "แนะนำหน่อย",
        "ดูของ",
        "สวยๆ",
        "ดีๆ"
    ]
    
    def is_vague(self, user_input):
        """Check if request is too vague"""
        
        # Count specific information
        has_occasion = any(occ in user_input for occ in ["ทำงาน", "งาน", "แต่ง", "เที่ยว"])
        has_budget = any(char.isdigit() for char in user_input)
        has_style = any(style in user_input for style in ["สบาย", "เป็นทางการ", "เก๋"])
        has_color = any(color in user_input for color in ["ดำ", "ขาว", "แดง", "น้ำเงิน"])
        
        specifics_count = sum([has_occasion, has_budget, has_style, has_color])
        
        # If less than 2 specifics, it's vague
        return specifics_count < 2
    
    def clarify(self, user_input):
        """Generate clarification questions"""
        
        questions = ["อยากช่วยเลยค่ะ! แต่ขอถามหน่อยนะ:"]
        
        # Ask about what's missing
        if "ทำงาน" not in user_input and "งาน" not in user_input:
            questions.append("- จะใส่ไปไหนคะ? (ทำงาน/เที่ยว/งานพิเศษ?)")
        
        if not any(char.isdigit() for char in user_input):
            questions.append("- งบประมาณเท่าไหร่พอดีคะ?")
        
        if not any(style in user_input for style in ["สบาย", "เป็นทางการ", "เก๋"]):
            questions.append("- ชอบสไตล์แบบไหนคะ? (สบายๆ/เป็นทางการ/เก๋ๆ?)")
        
        questions.append("\nแล้วจะหาให้เลยนะ! 😊")
        
        return "\n".join(questions)
```

### **Conflict Detector**

```python
class ConflictDetector:
    """
    Detect and resolve conflicting requirements
    """
    
    def detect_conflicts(self, requirements):
        """Find conflicts in user requirements"""
        
        conflicts = []
        
        # Budget vs Designer conflict
        if requirements.get("budget", 0) < 2000 and requirements.get("quality") == "designer":
            conflicts.append({
                "type": "budget_quality",
                "message": "งบ < 2000 แต่ต้องการ designer",
                "resolution": "suggest_affordable_brands"
            })
        
        # Formal vs Comfortable conflict
        if requirements.get("occasion") == "interview" and requirements.get("priority") == "comfort":
            conflicts.append({
                "type": "occasion_comfort",
                "message": "สัมภาษณ์งาน แต่ต้องการสบายมาก",
                "resolution": "balance_both"
            })
        
        # Multiple occasions conflict
        if len(requirements.get("occasions", [])) > 2:
            conflicts.append({
                "type": "too_many_occasions",
                "message": "ต้องการใส่ได้หลายโอกาสเกินไป",
                "resolution": "suggest_versatile"
            })
        
        return conflicts
    
    def resolve_conflict(self, conflict_type, requirements):
        """Generate resolution message"""
        
        if conflict_type == "budget_quality":
            return f"""เข้าใจค่ะว่าอยากได้คุณภาพดีๆ! 👍

ตามงบ {requirements['budget']} บาท:
- Designer จริงๆ จะยาก (เริ่ม 2,000+ บาท)
- แต่มีแบรนด์ดีๆ ราคาเอื้อมถึงได้นะ!

แนะนำ:
✅ Uniqlo, CPS - คุณภาพดี ราคาโอเค
✅ Central Store Brand - ราคาดี สไตล์ดี
✅ รอเซล - designer ลดราคาได้ดี!

ลองดูมั้ยคะ? 😊"""
```

```python
            return self.format_outfit_response(outfit, user_profile)
        
        return None
    
    def allocate_budget(self, total_budget, components):
        """Smart budget allocation across components"""
        
        # Typical splits
        ALLOCATIONS = {
            "dress": 0.50,
            "top": 0.30,
            "bottom": 0.30,
            "shoes": 0.25,
            "bag": 0.15,
            "blazer": 0.35,
            "accessories": 0.10
        }
        
        budget_split = {}
        required_components = [c for c in components if "_optional" not in c]
        
        for component in required_components:
            allocation = ALLOCATIONS.get(component, 0.20)
            budget_split[component] = total_budget * allocation
        
        return budget_split
    
    def validate_outfit(self, outfit):
        """Validate outfit coordination"""
        
        # Color coordination
        if not self.colors_match(outfit):
            return False
        
        # Style consistency
        if not self.styles_consistent(outfit):
            return False
        
        # Occasion appropriate
        if not self.occasion_appropriate(outfit):
            return False
        
        return True
    
    def colors_match(self, outfit):
        """Check if colors coordinate"""
        
        colors = [item.color for item in outfit.values()]
        
        # Max 3 colors
        if len(set(colors)) > 3:
            return False
        
        # Check for clashing colors
        clashing_pairs = [
            ("red", "pink"),
            ("orange", "red"),
            ("purple", "brown")
        ]
        
        for color1, color2 in clashing_pairs:
            if color1 in colors and color2 in colors:
                return False
        
        return True
```

```python
class ColorMatcher:
    """
    Color coordination logic
    """
    
    SAFE_COMBINATIONS = [
        # Neutral + Neutral
        ("black", "white"),
        ("black", "grey"),
        ("navy", "white"),
        ("beige", "white"),
        ("grey", "white"),
        
        # Neutral + Color
        ("black", "red"),
        ("navy", "white"),
        ("beige", "blue"),
        ("grey", "yellow"),
        ("white", "*"),  # White goes with everything
        
        # Complementary colors
        ("blue", "orange"),
        ("purple", "yellow"),
        ("red", "green"),
        
        # Analogous colors
        ("blue", "green"),
        ("red", "orange"),
        ("purple", "pink")
    ]
    
    AVOID_COMBINATIONS = [
        ("red", "pink"),
        ("orange", "red"),
        ("brown", "black"),
        ("navy", "black"),  # Too similar
        ("purple", "brown")
    ]
    
    def colors_coordinate(self, color1, color2):
        """Check if two colors coordinate"""
        
        # Check safe combinations
        if (color1, color2) in self.SAFE_COMBINATIONS:
            return True
        if (color2, color1) in self.SAFE_COMBINATIONS:
            return True
        
        # Check avoid combinations
        if (color1, color2) in self.AVOID_COMBINATIONS:
            return False
        if (color2, color1) in self.AVOID_COMBINATIONS:
            return False
        
        # Neutral colors generally work
        neutrals = ["black", "white", "grey", "beige", "navy", "cream"]
        if color1 in neutrals or color2 in neutrals:
            return True
        
        return False
```

```python
class ConversationContext:

        }
    
    def update(self, turn_data):
        """Update context with new turn information"""
        
        self.context["turn"] += 1
        
        # Extract entities from user input
        if "occasion" in turn_data:
            self.context["current_occasion"] = turn_data["occasion"]
        
        if "budget" in turn_data:
            self.context["current_budget"] = turn_data["budget"]
        
        # Track items shown
        if "items_shown" in turn_data:
            self.context["items_shown"].extend(turn_data["items_shown"])
        
        # Detect conversation stage
        self.context["conversation_stage"] = self.detect_stage()
    
    def resolve_reference(self, reference):
        """Resolve references like 'อันแรก', 'สีนี้'"""
        
        if reference in ["อันแรก", "ตัวแรก", "อันนั้น"]:
            # Refers to first item shown
            if self.context["items_shown"]:
                return self.context["items_shown"][0]
        
        elif reference in ["อันสอง", "ตัวสอง"]:
            if len(self.context["items_shown"]) >= 2:
                return self.context["items_shown"][1]
        
        elif reference in ["อันนี้", "ตัวนี้", "สีนี้"]:
            # Refers to current item being discussed
            return self.context["current_item"]
        
        elif reference in ["อันที่ชอบ", "อันที่ว่า"]:
            # Refers to item user expressed interest in
            for item in self.context["items_discussed"]:
                if item.get("user_interested"):
                    return item
        
        return None
    
    def detect_stage(self):
        """Detect current conversation stage"""
        
        if not self.context["current_occasion"]:
            return "discovery"
        
        elif not self.context["items_shown"]:
            return "searching"
        
        elif len(self.context["items_discussed"]) > 0:
            return "refinement"
        
        elif self.context.get("user_ready_to_buy"):
            return "purchase"
        
        return "exploration"
```

```python
            "next_stage": "recommendation"
        },

            "goal": "Show relevant options",
            "actions": ["search_products", "show_3-5_options"],
            "next_stage": "refinement"
        },
        "refinement": {
            "goal": "Narrow down choices",
            "questions": ["which_liked", "what_wrong"],
            "next_stage": "decision"
        },
        "decision": {
            "goal": "Help choose final item",
            "actions": ["compare", "answer_questions"],
            "next_stage": "purchase"
        },
        "purchase": {
            "goal": "Facilitate purchase",
            "actions": ["check_stock", "provide_link"],
            "next_stage": "complete"
        }
    }
    
    def get_next_action(self, current_stage, context):
        """Determine next action based on stage"""
        
        stage_info = self.STAGES[current_stage]
        
        if "questions" in stage_info:
            # Check what's missing
            missing = [
                q for q in stage_info["questions"]
                if not context.get(q)
            ]
            
            if missing:
                return ("ask_question", missing[0])
        
        if "actions" in stage_info:
            return ("perform_action", stage_info["actions"][0])
        
        # Move to next stage
        return ("change_stage", stage_info["next_stage"])
```
